# 第 12 章 #VMEXIT 分发器与五大退出码

## 12.1 分发表

```c
typedef int (*exit_fn_t)(vcpu_t *vc);

static const struct {
    u32 code;
    exit_fn_t fn;
    const char *name;
} g_exit_tab[] = {
    { 0x06A, ex_cpuid,    "CPUID"    },
    { 0x070, ex_hlt,      "HLT"      },
    { 0x073, ex_ioio,     "IOIO"     },
    { 0x074, ex_msr,      "MSR"      },
    { 0x077, ex_shutdown, "SHUTDOWN" },
    { 0x400, ex_npf,      "NPF"      },
    { 0x079, ex_vmmcall,  "VMMCALL"  },
};

int vmexit_dispatch(vcpu_t *vc, u32 code)
{
    for (u32 i = 0; i < sizeof(g_exit_tab)/sizeof(g_exit_tab[0]); i++) {
        if (g_exit_tab[i].code == code) {
            int r = g_exit_tab[i].fn(vc);
            if (r < 0) {
                serial_puts("EXIT-ERR "); serial_puts(g_exit_tab[i].name);
                serial_puts(" rip=0x"); serial_hex(vc->last_rip); serial_putc('\n');
            }
            return r;
        }
    }
    /* 兜底：未实现的退出码 */
    serial_puts("EXIT-UNHANDLED code=0x"); serial_hex(code);
    serial_puts(" info1=0x"); serial_hex((u32)vmcb_rd64(vc->vmcb, VMCB_OFF_EXIT_INFO1));
    serial_puts(" info2=0x"); serial_hex((u32)vmcb_rd64(vc->vmcb, VMCB_OFF_EXIT_INFO2));
    serial_puts(" rip=0x");   serial_hex(vc->last_rip);
    serial_putc('\n');
    vc->state = VCPU_STATE_FAULTED;
    return -EIO;
}
```

顺序查找 7 项的性能不是问题（每次退出一次线性扫描 7 次比较）。**不要**为了"优雅"改成哈希或函数指针数组——那会让"未知退出码"的处理变复杂，而未知退出码的处理才是这张表的核心价值。

## 12.2 CPUID（`0x06A`）

CPUID 是**唯一必须虚拟化而不能直通的**指令：直通会把宿主的 CPU 型号、拓扑、特性全暴露给 guest，导致 guest 内核选择错误的代码路径（最经典的后果：guest 看到 SVG/AVIC 位，以为自己也能虚拟化）。

```c
static int ex_cpuid(vcpu_t *vc)
{
    u32 leaf = vmcb_rd32(vc->vmcb, VMCB_OFF_RAX);          /* EAX */
    u32 sub  = vmcb_rd32(vc->vmcb, VMCB_OFF_RAX + 8);      /* ECX（子叶） */
    u32 a = 0, b = 0, c = 0, d = 0;

    switch (leaf) {
    case 0x00000000:                        /* vendor + max basic leaf */
        a = 0x0000000D;                     /* 我们愿意"声称"支持到 0xD */
        b = 0x756E7563;                     /* 'c','u','n','u' —— 见下 */
        d = 0x4D4D5673;                     /* 's','V','M','M'（示意，需按字节序核对） */
        c = 0x00000000;
        break;
    case 0x00000001:
        a = 0x00000600;                     /* family/model 归一为 P6 级 */
        b = 0;                              /* 不暴露拓扑细节（单核） */
        c = 0x00000000;
        d = CPUID1_EDX_BASE;                /* FPU|TSC|MSR|PAE|PSE|CX8|APIC|... */
        break;
    case 0x80000000:                        /* max extended leaf */
        a = 0x80000008;
        b = c = d = 0;
        break;
    case 0x80000001:
        /* 关键：ECX[2]（SVM）必须清 0，否则 guest 会以为自己能跑 VMRUN；
         * EDX[29]（LM）也必须清 0，cunos 不支持长模式 guest。 */
        a = 0x00000600; b = 0; c = 0;
        d = 0;
        if (vc->vm->cpu_pae_ok) d |= (1u << 6);      /* PAE：阶段 3 才置 1 */
        break;
    case 0x8000000A:                        /* SVM 特性：全部返回 0 */
        a = b = c = d = 0;
        break;
    case 0x80000008:
        a = 0x00003020;                     /* 物理地址 32 位、虚拟地址 32 位 */
        b = c = d = 0;
        break;
    default:
        a = b = c = d = 0;                  /* 未知叶一律返回 0，绝不直通 */
        break;
    }

    vmcb_wr32(vc->vmcb, VMCB_OFF_RAX,     a);
    vmcb_wr32(vc->vmcb, VMCB_OFF_RAX + 4, 0);
    vmcb_wr32(vc->vmcb, VMCB_OFF_RAX + 8, b);   /* EBX */
    vmcb_wr32(vc->vmcb, VMCB_OFF_RAX + 12, 0);
    /* ECX/EDX 在 VMCB 里的偏移与 RAX 组不连续，必须查表后写（【需核对】RCX/RDX 槽位） */
    vmcb_wr32(vc->vmcb, VMCB_OFF_RCX, c);
    vmcb_wr32(vc->vmcb, VMCB_OFF_RDX, d);

    vc->cpuid_count++;
    /* CPUID 不改变 RIP —— 等等：CPUID 是 2 字节指令（0F A2），
     * 但它由硬件"以当前 RIP 重新执行"的方式退出，因此宿主必须前推 2 */
    vmcb_wr64(vc->vmcb, VMCB_OFF_RIP, vmcb_rd64(vc->vmcb, VMCB_OFF_RIP) + 2);
    return 0;
}
```

三点必须强调：

1. **CPUID 的 RIP 必须手工前推 2**。这是最容易忘的一步，忘了就是"guest 无限次执行同一条 CPUID"，现象是主机卡死（退出循环不推进），而串口只会刷满你的计数器日志。
2. **vendor 字符串的字节序**：CPUID 返回的 EBX/EDX/ECX 拼起来是"内存小端序的 ASCII"。写 `0x756E7563` 时看到的是 `cunu`，容易写反。**建议直接构造常数并打印验证**，别靠心算。
3. **一叶一会话**：`0x80000001` 的 ECX 是"扩展特性"，与 EDX 的"基本特性"分属两个寄存器。清 SVM 位只清 ECX[2]，别把 EDX 也清零——EDX 里有 `SYSCALL`/`NX`/`RDTSCP`，清掉之后 Debian 内核的行为会变得很奇怪。

`VMCB_OFF_RCX` / `VMCB_OFF_RDX` 需要在 VMCB 偏移表里补上（本文 9.5 未列全）。**这提醒一件事**：`vmcb.h` 的偏移表必须完整抄录一遍 APM 的 state save area 图，不能只抄"自己以为会用到"的字段——退出处理随时需要读写任意通用寄存器。

## 12.3 HLT（`0x070`）

```c
static int ex_hlt(vcpu_t *vc)
{
    vc->hlt_count++;

    if (vc->exit_on_hlt) {
        vc->state = VCPU_STATE_HALTED;
        vmcb_wr64(vc->vmcb, VMCB_OFF_RIP, vmcb_rd64(vc->vmcb, VMCB_OFF_RIP) + 1);
        return VCPU_EXIT_NORMAL;
    }

    /* 正常语义：guest 空闲，等下一个事件（PIT tick / 外部输入） */
    vc->state = VCPU_STATE_HALTED;
    vmcb_wr64(vc->vmcb, VMCB_OFF_RIP, vmcb_rd64(vc->vmcb, VMCB_OFF_RIP) + 1);

    /* 让出：回到宿主，等下次 vm_run 或事件到达再继续。
     * 阶段 2 用"忙等 + 检查 pending"，阶段 3 用内核线程 + 条件变量。 */
    while (!vcpu_has_pending(vc)) {
        vmm_idle_wait();                    /* 见下：不能真的 hlt 宿主 */
    }
    vc->state = VCPU_STATE_READY;
    return 0;
}
```

`vmm_idle_wait()` 的实现要小心：**宿主绝对不能执行 `hlt` 指令**。因为宿主 IF=0 且 PIC 未初始化，`hlt` 之后没有任何中断能唤醒它——机器彻底挂死。阶段 2 的正确做法是"忙等 + 定期检查设备 tick"，或者干脆让宿主在 `hlt` 后直接返回 `vm_run()`、由上层决定何时再进入。

```c
static void vmm_idle_wait(void)
{
    /* 忙等一小段：用 pause 降低功耗，同时给设备 tick 机会 */
    for (u32 i = 0; i < 1000; i++) {
        __asm__ volatile("pause");
    }
    u64 now = rdtsc();
    for (u32 i = 0; i < g_cur_vm->dev_cnt; i++)
        if (g_cur_vm->devs[i]->ops->tick)
            g_cur_vm->devs[i]->ops->tick(g_cur_vm->devs[i], now);
}
```

## 12.4 IOIO（`0x073`）

```c
static int ex_ioio(vcpu_t *vc)
{
    u64 info1 = vmcb_rd64(vc->vmcb, VMCB_OFF_EXIT_INFO1);
    u16 port  = (u16)(info1 & 0xFFFFu);
    u32 is_in = (info1 >> 16) & 1u;          /* 1 = IN，0 = OUT */
    u32 width = (u32)((info1 >> 17) & 3u);   /* 【需核对】编码含义，以下按 0=1B,1=2B,3=4B 处理 */
    static const u32 wtab[4] = { 1, 2, 4, 8 };
    u32 bytes = wtab[width & 3];
    u32 val = 0;

    vc->io_count++;

    if (is_in) {
        vdev_t *d = vdev_find_io(vc->vm, port);
        if (!d) { inject_gp(vc); return 0; }
        if (d->ops->io_read(d, port, bytes, &val) < 0) { inject_gp(vc); return 0; }
        vmcb_wr32(vc->vmcb, VMCB_OFF_RAX, val);          /* 返回值写 EAX（IN 指令总是回 AX/EAX） */
    } else {
        val = vmcb_rd32(vc->vmcb, VMCB_OFF_RAX) & (bytes == 4 ? 0xFFFFFFFFu : ((1u << (bytes*8)) - 1));
        vdev_t *d = vdev_find_io(vc->vm, port);
        if (!d) { inject_gp(vc); return 0; }
        if (d->ops->io_write(d, port, bytes, val) < 0) { inject_gp(vc); return 0; }
    }

    /* 指令长度：0xE4..0xEF 家族是 2 字节（含 imm8）或 1 字节（端口在 DX）；
     * 保险做法是用 5.3 的"读 guest 指令字节"来精确前推。 */
    u32 len = io_insn_len(vc);
    vmcb_wr64(vc->vmcb, VMCB_OFF_RIP, vmcb_rd64(vc->vmcb, VMCB_OFF_RIP) + len);
    return 0;
}
```

工程细节：

- **宽度的编码在不同 APM 版本里存在歧义**，所以函数里保留了 `io_insn_len()` / `io_width_from_insn()`（5.3）这条"用指令字节反解"的旁路。阶段 2 上线时两者并行断言，稳定后再删。
- **字符串 IO（`ins`/`outs`，`rep` 前缀）** 不在阶段 1-2 范围。Debian 内核的 `inb/outb` 都是单次指令，只有 `rep insb` 用于读取磁盘数据（PIO 模式）——那时我们会用 virtio-blk 而不是 ATA PIO，所以可以长期不支持字符串 IO。若必须支持，`exit_info_1` 里另有"是否为字符串指令"的标志位，需要在每个迭代上处理 `RCX/RSI/RDI` 的推进。
- **IN 的返回值必须写 EAX 而不是 AX**：`in al, dx` 只改 AL，但写整个 EAX 的低 8 位与写 AX 的语义在高位上有差异（`in al` 保留 AH）。严谨做法是按 `bytes` 只覆盖对应字节：

```c
static void set_rax_partial(u8 *v, u32 bytes, u32 val)
{
    u32 old = vmcb_rd32(v, VMCB_OFF_RAX);
    if (bytes == 1)      val = (old & 0xFFFFFF00u) | (val & 0xFFu);
    else if (bytes == 2) val = (old & 0xFFFF0000u) | (val & 0xFFFFu);
    vmcb_wr32(v, VMCB_OFF_RAX, val);
    vmcb_wr32(v, VMCB_OFF_RAX + 4, 0);
}
```

## 12.5 MSR（`0x074`）

```c
static int ex_msr(vcpu_t *vc)
{
    u64 info1 = vmcb_rd64(vc->vmcb, VMCB_OFF_EXIT_INFO1);
    u32 is_wr = (u32)(info1 & 1u);                  /* 1 = WRMSR（【需核对】bit 位置） */
    u32 msr   = (u32)vmcb_rd64(vc->vmcb, VMCB_OFF_EXIT_INFO2);
    u64 val   = vcpu_get_msr_operand(vc, is_wr);    /* 写时取 EDX:EAX，读时忽略 */

    vc->msr_count++;

    switch (msr) {
    case 0xC0000080u:   /* EFER：只允许 guest 改 NXE 位，SVME/LME 一律拒绝 */
        if (is_wr) {
            if (val & (EFER_SVME | (1ull << 8))) {  /* 试图开 SVM 或 LME → #GP */
                inject_gp(vc); return 0;
            }
            vcpu_set_guest_efer(vc, val & ~(EFER_SVME));
        } else {
            vcpu_put_msr_operand(vc, vcpu_get_guest_efer(vc));
        }
        break;

    case 0x00000174u: case 0x00000175u: case 0x00000176u:   /* SYSENTER_CS/ESP/EIP */
        if (is_wr) vc->guest_sysenter[msr - 0x174] = val;
        else       vcpu_put_msr_operand(vc, vc->guest_sysenter[msr - 0x174]);
        break;

    case 0xC0000081u: case 0xC0000082u: case 0xC0000083u: case 0xC0000084u:
        /* STAR / LSTAR / CSTAR / SFMASK：写到 VMCB 的对应槽位
         * （这些值 VMRUN 时会自动装载到硬件，因此"存 VMCB"即"生效"）*/
        vcpu_set_guest_msr_slot(vc, msr, val);
        break;

    case 0x00000277u:   /* CR_PAT */
        vcpu_set_guest_pat(vc, val);
        break;

    default:
        /* 白名单之外：一律注入 #GP —— 绝不能转发给物理 MSR */
        inject_gp(vc);
        return 0;
    }

    /* RDMSR/WRMSR 是 2 字节指令（0F 30 / 0F 32） */
    vmcb_wr64(vc->vmcb, VMCB_OFF_RIP, vmcb_rd64(vc->vmcb, VMCB_OFF_RIP) + 2);
    return 0;
}
```

**MSR 的默认拒绝策略是安全边界**。如果对未知 MSR 直接 `wrmsr(msr, val)` 转发给物理 MSR，guest 只要写 `MSR_VM_HSAVE_PA`（`0xC0010117`）就能把宿主的 HSAVE 页改到 guest 自己的内存里，接下来 `VMRUN` 的宿主上下文人尽可改——这是**整机接管级别的漏洞**。MSRPM 全 1 + 白名单是实现层面的双保险。

第二道保险：**MSRPM 的位映射不是线性的**。MSR 空间被切成若干区间（`0x00000000-0x00001FFF`、`0x40000000-0x40001FFF`、`0xC0000000-0xC0001FFF` 等），每区间独立一段位图。【需核对：APM 中 MSRPM 分段与偏移计算公式】本文的做法是"全部置 1（全拦）+ 白名单逐个清位"，这样即使公式记错，也只是"多拦了本该直通的 MSR"（表现为 guest 慢、偶发 `#GP`），而不会"漏拦危险 MSR"。

## 12.6 SHUTDOWN（`0x077`）

```c
static int ex_shutdown(vcpu_t *vc)
{
    u32 reason = (u32)vmcb_rd64(vc->vmcb, VMCB_OFF_EXIT_INFO1);
    serial_puts("\n[guest SHUTDOWN] reason=0x"); serial_hex(reason);
    serial_puts(" rip=0x"); serial_hex(vc->last_rip);
    serial_puts(" cpl=");   serial_dec(*(u8 *)(vc->vmcb + VMCB_OFF_CPL));
    serial_putc('\n');

    /* 常见原因：三落（无 IDT 时收到异常）、CR0/CR4 非法组合、段加载失败 */
    serial_puts("hint: check guest IDT/GDT setup, CR0.PE vs CS attrib, stack validity\n");

    vc->vm->shutdown_reason = reason;
    vc->state = VCPU_STATE_SHUTDOWN;
    return VCPU_EXIT_SHUTDOWN;
}
```

`SHUTDOWN` 是**唯一会让我们看到"guest 内核自己崩了"的窗口**。它必须打印足够信息，否则 Stage 3 会陷入"Debian 黑屏，什么都不知道"的地狱。

## 12.7 两个额外退出码

**VMMCALL（`0x079`）——超调用通道。** 这是我们与 guest 之间的"私有 API"，用途：

| 超调用号 | 参数 | 作用 |
|---|---|---|
| `0x01` | `EBX` = 串口字符 | 直接打印（等价于 `out 0x3F8`，但一次退出打一行更快） |
| `0x02` | — | 返回 `EAX` = 已运行 tick 数 |
| `0x03` | — | 请求退出 `vm_run`（正常关机语义） |
| `0x04` | `EBX` = GPA，`ECX` = 长度 | 宿主把该段 guest 内存 dump 到串口（调试用，**仅在 debug 构建启用**） |

超调用的约定要写进 guest 侧头文件：

```c
/* guest 侧的用法（运行在 guest 里，被我们自己的 loader/驱动调用）*/
static inline u32 vmm_hc(u32 nr, u32 a1, u32 a2)
{
    u32 r;
    __asm__ volatile("vmcall" : "=a"(r) : "a"(nr), "b"(a1), "c"(a2));
    return r;
}
```

**NPF（`0x400`）**：见 6.4，这里不重复。它不在"五大"里，但是阶段 1 就必须实现的（`npt.c` 的 `npt_handle_npf`）。
