# 第 13 章 设备模型

## 13.1 统一约定与资源分配表

所有设备都必须遵守三条：

1. **端口/MMIO 范围全局唯一**，`vdev_init_all()` 里做重叠检测（重叠即 `-EBUSY`，打印两个设备名）。
2. **设备只操作自己的寄存器堆，不访问 guest 内存**（唯一例外：块设备读写 vring，见 13.6；那时它拿的是 GPA，通过 `ram_va()` 访问，且必须做边界检查）。
3. **中断通过"待注入队列"上报**，不直接改 VMCB（避免设备在奇怪时刻改 VMCB 导致状态不一致）。

资源分配表（阶段 2 的完整 PC 画像）：

| 设备 | 端口 / MMIO | 长度 | IRQ | 备注 |
|---|---|---|---|---|
| 8259A 主片 | `0x20`–`0x21` | 2 | — | 级联到 IRQ2 |
| 8259A 从片 | `0xA0`–`0xA1` | 2 | — | |
| 8254 PIT | `0x40`–`0x43` | 4 | IRQ0 | tick 源 |
| 8042 键盘 | `0x60`、`0x64` | 2 | IRQ1 | |
| 端口 0x61 | `0x61` | 1 | — | NMI 状态/扬声器门控 |
| CMOS/RTC | `0x70`–`0x71` | 2 | IRQ8 | 阶段 4 才做 |
| DMA 控制器 | `0x00`–`0x0F`、`0x80`–`0x8F` | 16+16 | — | 阶段 4；**先给"空设备"避免 guest 探测卡死** |
| 8250 COM1 | `0x3F8`–`0x3FF` | 8 | IRQ4 | 主要输出通道 |
| 8250 COM2 | `0x2F8`–`0x2FF` | 8 | IRQ3 | 可选 |

## 13.2 8259A 可编程中断控制器

PIC 是 Debian 内核最早期就会初始化的设备（`8259A` 的 ICW 序列是内核启动的必经之路）。它必须"看起来对"，否则内核在 `init_IRQ` 阶段就挂。

寄存器只有两个端口（主片 `0x20/0x21`），行为取决于内部状态机：

```c
typedef struct {
    u8  irr;            /* 中断请求寄存器（待处理） */
    u8  isr;            /* 服务中寄存器（已应答） */
    u8  imr;            /* 屏蔽寄存器（端口 base+1） */
    u8  vector_base;    /* ICW2 给出的基向量：主片 0x20，从片 0x28 */
    u8  elcr;           /* 触发模式：1=电平，0=边沿 */
    u8  expect_icw;     /* 正在等第几个 ICW（0=不在初始化，1..4=等 ICWn） */
    u8  icw_step;
    u8  read_isr;       /* OCW3：读 ISR 还是 IRR */
    u8  special_mask;   /* OCW3：特殊屏蔽模式 */
    u8  poll_mode;      /* OCW3：轮询模式 */
    int is_slave;
    int cascade_irq;    /* 从片挂在主片的 IRQ2 上 */
} pic_t;
```

写入路径的完整状态机（这是 PIC 实现的全部难点，写清了就没有别的坑）：

```c
static int pic_write(pic_t *p, u16 port, u32 val)
{
    u8 v = (u8)val;
    int is_cmd = ((port & 1) == 0);      /* 偶地址 = 命令口（A0=0） */

    if (v & 0x10) {                       /* bit4=1 → 这是 ICW1，进入初始化 */
        p->expect_icw = 1;
        p->imr = 0;
        p->isr = 0;
        p->irr = 0;
        p->elcr = 0;
        p->icw_step = 1;
        /* ICW1 bit0=1 表示需要 ICW4；bit1=1 表示单片（无从片） */
        p->need_icw4 = v & 1;
        p->single    = (v >> 1) & 1;
        return 0;
    }

    if (p->expect_icw) {                  /* 正在收 ICW2..ICW4（都写命令口） */
        p->icw_step++;
        if (p->icw_step == 2) {
            p->vector_base = v & 0xF8;    /* ICW2：基向量，低 3 位被忽略 */
        } else if (p->icw_step == 3) {
            /* ICW3：主片是级联位图，从片是级联号。本模型单从片 → 主片 bit2=1 */
        } else if (p->icw_step == 4) {
            p->expect_icw = 0;            /* ICW4：8086 模式等，忽略 */
        }
        return 0;
    }

    if (!is_cmd) {                        /* 奇地址 = 数据口：OCW1（中断屏蔽） */
        p->imr = v;
        return 0;
    }

    /* OCW2：EOI 与优先级轮转 */
    if (v & 0x18) {
        u8 op = (v >> 5) & 7;
        if (op == 1) {                    /* 非特定 EOI：清最高优先级 ISR 位 */
            p->isr &= (u8)(p->isr - 1);
        } else if (op == 3) {             /* 特定 EOI：清指定 ISR 位 */
            p->isr &= (u8)~(1u << (v & 7));
        }
        pic_update(p);                    /* 重新评估谁能打断 */
        return 0;
    }

    /* OCW3：读寄存器选择 / 轮询 / 特殊屏蔽 */
    if (v & 0x08) {
        p->read_isr  = v & 0x01;
        p->poll_mode = (v >> 2) & 1;
    }
    return 0;
}
```

读取路径（`0x20` 读 IRR/ISR，`0x21` 读 IMR）：

```c
static int pic_read(pic_t *p, u16 port, u32 *out)
{
    if ((port & 1) == 0) {
        u8 r = p->read_isr ? p->isr : p->irr;
        /* 轮询模式下，最高位=1 表示有请求，低 3 位给向量号 */
        if (p->poll_mode) r = 0x80 | 0x00;
        *out = r;
    } else {
        *out = p->imr;
    }
    return 0;
}
```

**中断注入的完整链条**（这是"PIT 到期 → guest 收到 IRQ0"的全过程）：

```
PIT tick 到期
  → vdev_pit 设置自己的 pending=1
  → vmexit 循环的 tick 检查点调用 pic_raise(pic, 0)
  → pic_raise: irr |= (1<<0); pic_update()
  → pic_update: 若 (irr & ~imr & ~isr) 非零且未在处理更高优先级
      → 最高优先级位 n → vector = vector_base + n
      → isr |= (1<<n); irr &= ~(1<<n)
      → vdev_pic 把 (vector, edge) 放进 vcpu 的待注入槽
  → vmexit 在下一次 VMRUN 之前调用 vcpu_deliver_pending()
  → vmcb 的 INT_CTL/INT_VECTOR/EXIT_INTINFO 填入 → VMRUN
  → guest 在 IF=1 时立刻收到中断，自动进入 guest 的 IDT[0x20]
```

对应代码：

```c
void vcpu_deliver_pending(vcpu_t *vc)
{
    if (!vc->pending_vector) return;

    /* 前置条件：guest 必须 IF=1，否则保持 pending 等待 */
    u64 rflags = vmcb_rd64(vc->vmcb, VMCB_OFF_RFLAGS);
    if (!(rflags & 0x200)) return;

    vmcb_wr32(vc->vmcb, VMCB_OFF_INT_CTL, 1);                     /* V_IRQ=1 */
    vmcb_wr32(vc->vmcb, VMCB_OFF_INT_VECTOR, vc->pending_vector); /* V_IRQ_VECTOR */
    /* 【需核对】V_IRQ 与 V_IRQ_VECTOR 的确切偏移与位宽 */
    vc->pending_vector = 0;
}
```

注意一个极易出错的地方：**中断注入时 guest 的 `RFLAGS.IF` 可能为 0（cli 中）**。此时不能强行注入（会造成"guest 在 cli 区间收到中断"这种违反语义的情况），必须保留 pending，等下次退出再检查。这也是"每次退出都检查 pending"的原因——退出点天然是注入机会。

## 13.3 8254 可编程间隔定时器

PIT 是时间基准，也是阶段 2 "guest 能连续运行"的关键。三通道，端口 `0x40/0x41/0x42`，控制口 `0x43`。

```c
typedef struct {
    u8  mode[3];          /* 通道模式 0..5 */
    u8  access[3];        /* 访问模式：1=只低字节,2=只高字节,3=先低后高 */
    u16 reload[3];        /* 重载值 */
    u16 counter[3];       /* 当前计数 */
    u8  latch[3];         /* 锁存命令：0=无,1=已锁存待读 */
    u16 latched[3];
    u8  write_hi[3];      /* 先低后高模式下，是否正在写高字节 */
    u8  read_hi[3];
    u64 period_tsc;       /* 通道 0 的周期，换算出 TSC 刻度 */
    u64 next_tick_tsc;
} pit_t;
```

控制字格式（`0x43`）：`bit7-6` 选择通道、`bit5-4` 访问模式、`bit3-1` 模式、`bit0` BCD。

```c
static int pit_write(pit_t *p, u16 port, u32 val)
{
    u8 v = (u8)val;

    if (port == 0x43) {                    /* 控制字 */
        u8 ch = (v >> 6) & 3;
        if (ch == 3) return 0;             /* 读回命令，阶段 3 再实现 */
        p->access[ch] = (v >> 4) & 3;
        p->mode[ch]   = (v >> 1) & 7;
        p->write_hi[ch] = 0;
        if (p->access[ch] == 1) p->write_hi[ch] = 1;   /* 只写低字节，视为已完成 */
        return 0;
    }

    u8 ch = (u8)(port - 0x40);
    switch (p->access[ch]) {
    case 1: p->reload[ch] = v; break;
    case 2: p->reload[ch] = (u16)(v << 8); break;
    case 3:
        if (!p->write_hi[ch]) { p->reload[ch] = (u16)(v & 0xFF); p->write_hi[ch] = 1; }
        else { p->reload[ch] = (u16)((p->reload[ch] & 0xFF) | (v << 8)); p->write_hi[ch] = 0; }
        break;
    }
    p->counter[ch] = p->reload[ch] ? p->reload[ch] : 0x10000;
    if (ch == 0) pit_recalc_period(p);
    return 0;
}
```

**周期换算**：PIT 输入时钟是 `1193182 Hz`（不是 1MHz，也不是 1193180——这个数的来源是 `14.31818MHz / 12`）。若 `reload = 0xFFFF`（Debian 内核通常用这种方式得到最大 18.2Hz 的 tick），

```c
static void pit_recalc_period(pit_t *p)
{
    u32 hz = 1193182u;
    u32 div = p->reload[0] ? p->reload[0] : 65536u;
    /* 周期（纳秒）→ TSC 刻度，需要 tsc_per_ns，在 vmm_init 用 PIT/PM 校准 */
    u64 ns = (u64)div * 1000000000ull / hz;
    p->period_tsc = ns * g_tsc_per_ns;
}
```

`g_tsc_per_ns` 的校准方法（无 `cpuid 0x15` 时的兜底）：在宿主里用 PIT 通道 2 的一次性模式跑一个已知时长（例如 `reload=0xFFFF` 的 55ms），前后读 TSC，相除得比例。**这个值决定 guest 时间快慢，必须实测，不能猜**。

`tick()` 实现：

```c
static int pit_tick(vdev_t *d, u64 now)
{
    pit_t *p = d->priv;
    if (!p->period_tsc) return 0;
    while (now >= p->next_tick_tsc) {
        p->next_tick_tsc += p->period_tsc;
        if (p->mode[0] != 0) {                 /* 模式 0（单次）只响一次 */
            pic_raise(d->vm->devs[PIC_IDX], 0); /* IRQ0 */
        }
    }
    return 0;
}
```

## 13.4 8250 UART

`0x3F8`–`0x3FF` 八个寄存器，靠 `LCR.DLAB` 复用到 DLL/DLM：

| 偏移 | DLAB=0 | DLAB=1 |
|---|---|---|
| +0 | RBR（读）/ THR（写） | DLL（波特率低字节） |
| +1 | IER | DLM（波特率高字节） |
| +2 | IIR（读）/ FCR（写） | — |
| +3 | LCR | — |
| +4 | MCR | — |
| +5 | LSR | — |
| +6 | MSR | — |
| +7 | SCR（暂存） | — |

```c
typedef struct {
    u8  dll, dlm, ier, fcr, lcr, mcr, scr;
    u8  lsr;              /* 恒为 THRE|TEMT|DR 就绪的状态组合 */
    u8  msr;
    u8  rx_buf[16];
    u32 rx_head, rx_tail;
    u8  tx_shift;         /* 发送移位寄存器（仅用于统计） */
} uart_t;
```

关键实现点：

```c
static int uart_write(vdev_t *d, u16 port, u32 val)
{
    uart_t *u = d->priv;
    u8 v = (u8)val;
    u32 off = port - d->io_base;

    if (u->lcr & 0x80) {                          /* DLAB=1：波特率除数 */
        if (off == 0) u->dll = v; else if (off == 1) u->dlm = v;
        return 0;
    }

    switch (off) {
    case 0:                                       /* THR：真正的输出 */
        serial_putc(v);                           /* 直通宿主串口：阶段 1 的"现象" */
        u->lsr |= 0x40;                           /* TEMT 置位（立即发完） */
        if (u->ier & 0x02) pic_raise(d->vm->devs[PIC_IDX], 4);  /* THRE 中断 */
        return 0;
    case 1: u->ier = v; return 0;
    case 2: u->fcr = v; return 0;
    case 3: u->lcr = v; return 0;
    case 4: u->mcr = v;
            /* MCR.bit4 = 内部回环：Debian 的串口探测会做这个动作，
             * 必须让 LSR/MSR 在回环下"看起来通"，否则它判定串口不存在 */
            return 0;
    case 7: u->scr = v; return 0;
    }
    return -1;                                    /* 未实现 → 注入 #GP */
}
```

`LSR` 的严格取值：`bit0=DR`（有数据）、`bit5=THRE`（发送保持寄存器空）、`bit6=TEMT`（发送移位寄存器空）。我们让 `LSR = 0x60`（THRE|TEMT 常置），`DR` 只在接收缓冲非空时置位。

**为什么 UART 值得这么细？** 因为阶段 1 的全部"现象"都从这里出来。如果 `LSR` 写错（例如写成 0），guest 的轮询 `while (!(inb(0x3FD) & 0x20));` 会永远不成立——现象是"进入 guest 后卡住，一个字符都不打印"，而 `exit_count` 却在飞涨（因为 `in` 指令不断产生 IOIO 退出）。看到"`io` 计数暴涨但无输出"，就是 `LSR` 的问题。

## 13.5 PCI 配置空间（`0xCF8` / `0xCFC`）

guest 看到的总线必须是**虚拟的**。做法是自建一张设备表，配置空间按需生成：

```c
typedef struct {
    u8   bus, dev, fn;
    u16 vendor_id, device_id;
    u8   class_code, subclass, prog_if, header_type;
    u8   irq_line;
    struct { u32 base; u32 size; u32 is_io; } bar[6];
    u32  bar_written[6];       /* guest 写 BAR 的探测过程要"吸收"掉 */
    u8   cfg[256];             /* 静态部分快照 */
} vpci_dev_t;
```

`0xCF8` 是地址口（`bit31=Enable`，`bit23-16=bus`，`bit15-11=dev`，`bit10-8=fn`，`bit7-2=reg`），`0xCFC` 是数据口（4 字节对齐）。

```c
static u32 pci_read_cfg(u32 cf8, u32 off)
{
    u32 bus = (cf8 >> 16) & 0xFF, dev = (cf8 >> 11) & 0x1F, fn = (cf8 >> 8) & 7;
    u32 reg = (cf8 & 0xFC) + off;
    vpci_dev_t *d = vpci_find(bus, dev, fn);

    if (!d) {
        /* 不存在的设备返回全 1（真实硬件语义，guest 靠这个判空） */
        return 0xFFFFFFFFu;
    }
    return *(u32 *)(d->cfg + reg);
}
```

BAR 的探测必须"像真硬件"：guest 写 `0xFFFFFFFF` 到 BAR，读回得到"大小掩码"，再写真实地址。我们的实现里，写全 1 时返回 `~(size-1)`，写普通值时记录并回读同值。

```c
static void pci_write_bar(vpci_dev_t *d, u32 idx, u32 val)
{
    if (val == 0xFFFFFFFFu) {
        u32 sz = d->bar[idx].size;
        *(u32 *)(d->cfg + 0x10 + idx * 4) = d->bar[idx].is_io
            ? (~(sz - 1) | 1u) : ~(sz - 1);
        return;
    }
    d->bar[idx].base = val & ~0xFu;
    *(u32 *)(d->cfg + 0x10 + idx * 4) = d->bar[idx].base | (d->bar[idx].is_io ? 1u : 0u);
}
```

## 13.6 块设备：ATA 模拟 vs virtio-blk

这是整个设备层最重要的取舍点。两条路的对比：

| 维度 | ATA/IDE PIO 模拟 | virtio-blk（legacy PCI） |
|---|---|---|
| guest 侧依赖 | 内核内置 `ata_piix`，**无需额外驱动** | 需要 `virtio_pci` + `virtio_blk` 模块；**必须进 initrd** |
| 我们实现量 | 端口命令块 + PIO 数据搬运 + 时序状态机 ≈ 300 行 | vring + notify + 描述符链 ≈ 300 行 |
| 每次 IO 的退出次数 | **512 字节 = 256 次 `rep insw` 迭代**（若不支持字符串 IO 则是 256 次退出） | **1 次 notify 退出**，之后批量读写 |
| 性能 | 极差（每次读扇区数百次退出） | 好（一次退出搬 4KB+） |
| 调试难度 | 状态机时序（DRQ/BSY）容易写错 | 环形队列索引（idx 回绕）容易写错 |
| 适合阶段 | 阶段 3 的"能起来"验证 | 阶段 3.5 / 4 的"能用" |

**推荐路线：两条都写，ATA 先行，virtio 紧随。** 理由很功利：ATA 让阶段 3 的 Debian 快速进入 shell（不用折腾 initrd），virtio 让阶段 4 有可接受的 IO 性能（`apt` 才不会慢到无法忍受）。

virtio-blk legacy 的关键结构（严格按 virtio 1.0 的 legacy 接口，字段顺序不可改）：

```c
/* 描述符：16 字节，必须 16 字节对齐 */
struct vring_desc {
    u64 addr;      /* GPA */
    u32 len;
    u16 flags;     /* 1=NEXT, 2=WRITE, 4=INDIRECT */
    u16 next;
};

/* 可用环：guest 写，host 读 */
struct vring_avail {
    u16 flags;
    u16 idx;
    u16 ring[];    /* 长度 = queue_size */
};

/* 已用环元素：8 字节 */
struct vring_used_elem { u32 id; u32 len; };

/* 已用环：host 写，guest 读 */
struct vring_used {
    u16 flags;
    u16 idx;
    struct vring_used_elem ring[];
};

/* 请求头：virtio-blk 的三段式描述符链 = [头][数据][状态] */
struct virtio_blk_req_hdr {
    u32 type;      /* 0=IN(读), 1=OUT(写), 4=FLUSH, 8=GET_ID */
    u32 reserved;
    u64 sector;    /* 以 512 字节为单位 */
};
#define VIRTIO_BLK_S_OK     0
#define VIRTIO_BLK_S_IOERR  1
#define VIRTIO_BLK_S_UNSUPP 2
```

legacy PCI 的 IO BAR 寄存器映射（**偏移必须准**）：

| 偏移 | 名称 | 宽 |
|---|---|---|
| `0x00` | device features | 4 |
| `0x04` | guest features | 4 |
| `0x08` | queue address（PFN，即 `vring_pa >> 12`） | 4 |
| `0x0C` | queue size（只读，我们给 128） | 2 |
| `0x0E` | queue select | 2 |
| `0x10` | queue notify（写即触发） | 2 |
| `0x12` | device status | 1 |
| `0x13` | ISR（读即清） | 1 |
| `0x14`+ | device config（blk：capacity 8 字节） | — |

notify 处理的骨架：

```c
static int vblk_notify(vdev_t *d, u16 qidx)
{
    vblk_t *b = d->priv;
    struct vring_avail *av = (struct vring_avail *)ram_va(b->vring_pa + b->avail_off);
    struct vring_used *us = (struct vring_used *)ram_va(b->vring_pa + b->used_off);

    while (b->last_avail != av->idx) {
        u16 head = av->ring[b->last_avail % b->qsize];
        /* 走描述符链：头(16B) → 数据(len) → 状态(1B) */
        struct vring_desc *desc = (struct vring_desc *)ram_va(b->vring_pa);
        u32 total = vblk_do_request(b, &desc[head]);   /* 真正读写镜像 */
        us->ring[us->idx % b->qsize].id  = head;
        us->ring[us->idx % b->qsize].len = total;
        us->idx++;
        b->last_avail++;
    }
    pic_raise(d->vm->devs[PIC_IDX], d->irq);           /* 完成后中断 */
    return 0;
}
```

**边界检查必须做**：`desc->addr` 与 `desc->len` 完全由 guest 控制。如果不检查，guest 可以让 `addr = 0xFFFFFFF0, len = 0x1000`，宿主 `ram_va()` 之后就越界读写了宿主内存——这是**虚拟机逃逸**。检查方式：

```c
static int gpa_range_ok(vm_t *vm, u64 gpa, u64 len)
{
    if (gpa > 0xFFFFFFFFull || len > 0xFFFFFFFFull) return 0;
    if (gpa + len < gpa) return 0;                       /* 溢出 */
    if (gpa + len > vm->ram_size) return 0;              /* 超出 guest RAM */
    return 1;
}
```

## 13.7 VGA / VBE 帧缓冲

阶段 2 只需要"能显示文字"（VGA 文本模式 0xB8000 的 80×25）。阶段 4 才需要图形模式。

Bochs VBE 兼容接口（`0x1CE/0x1CF`）是**开销最小的图形方案**，因为 QEMU/Bochs 的 VBE 扩展已被大量 guest 驱动（`bochs-drm`、`vesafb`）支持：

| 索引（写 `0x1CE`） | 含义 | 端口 |
|---|---|---|
| `0x00` | VBE_DISPI_INDEX_ID | `0x1CF` 数据 |
| `0x01` | XRES | |
| `0x02` | YRES | |
| `0x03` | BPP | |
| `0x04` | ENABLE（bit0=启用，bit6=线性帧缓冲） | |
| `0x06` | VIRT_WIDTH | |
| `0x07` | VIRT_HEIGHT | |
| `0x08` | X_OFFSET | |
| `0x09` | Y_OFFSET | |

```c
static int vbe_write(vdev_t *d, u16 port, u32 val)
{
    vbe_t *v = d->priv;
    if (port == 0x1CE) { v->index = (u16)val; return 0; }

    switch (v->index) {
    case 0x01: v->xres = (u16)val; break;
    case 0x02: v->yres = (u16)val; break;
    case 0x03: v->bpp  = (u16)val; break;
    case 0x04:
        v->enable = (u16)val;
        if (v->enable & 1) vbe_apply_mode(v);      /* 分配/调整影子帧缓冲 */
        break;
    }
    return 0;
}
```

影子帧缓冲的 GPA 固定为 `0xF0000000`（见第 14 章），大小按 `xres*yres*bpp/8` 计算，向上取 2MB 对齐。**它在 NPT 里不映射**——这样 guest 每次写像素都会 NPF 退出。这句话听起来很吓人（每次写像素都退出？），但实际不会：我们**第一次 NPF 时就把整个 4KB 页映射到影子缓冲的对应物理页**（`ex_npf` 里的"直通映射"分支），之后 guest 直接写、无退出。

```c
/* NPF 命中帧缓冲区域：把 GPA 映射到影子缓冲的 HPA，一次解决 4KB */
static int vbe_map_page(vbe_t *v, u32 gpa, u32 *out_pa)
{
    if (gpa < VBE_GPA_BASE || gpa >= VBE_GPA_BASE + v->fb_size) return -1;
    u32 off = gpa - VBE_GPA_BASE;
    *out_pa = v->fb_hpa_base + off;
    return 0;    /* 返回 0 表示"已映射，重试指令" */
}
```

## 13.8 8042 键盘控制器

`0x60`（数据）、`0x64`（状态/命令）。扫描码集合 1（XT 码）。Debian 在 PS/2 不可用时会退化到串口控制台，所以键盘是"可选但强烈建议"（阶段 4 才有窗口交互需求）。

```c
typedef struct {
    u8 out_buf;      /* 待 guest 读的扫描码 */
    u8 has_data;     /* 0x64 bit0 */
    u8 cmd;          /* 最近一条命令 */
    u8 kbd_enable;
} kbd_t;
```

宿主把按下的键翻译成扫描码后：

```c
void kbd_inject_scancode(vdev_t *d, u8 scan, u8 is_break)
{
    kbd_t *k = d->priv;
    if (!k->kbd_enable) return;
    k->out_buf = is_break ? (u8)(scan | 0x80) : scan;
    k->has_data = 1;
    pic_raise(d->vm->devs[PIC_IDX], 1);           /* IRQ1 */
}
```

## 13.9 两条实现路径的取舍（完整 PC 模拟 vs virtio 半虚拟化）

| 维度 | 完整 PC 模拟 | virtio 半虚拟化 |
|---|---|---|
| guest 兼容性 | 任何 i386 系统开箱即用（含 DOS、Win95 级的怪脾气系统） | 需要 guest 有驱动（Debian 有，但要在 initrd 里） |
| 实现工作量 | 大：PIC/PIT/DMA/IDE/VGA/RTC/PCI 全要"像真的"，时序细节无底洞 | 小：一个 notify 通路 + 显式协议 |
| 性能 | 差（大量退出与 PIO 搬运） | 好（批量、可减少退出） |
| 可调试性 | 差（guest 侧不知道我们在模拟，错误现象模糊） | 好（协议显式，可以打日志到"第几个请求"） |
| 失败模式 | "内核启动到某处挂住" | "驱动报 timeout"（有明确错误串） |
| 我们的选择 | **阶段 2-3 走这条路**（PIC/PIT/UART/VGA 文本/ATA） | **阶段 3.5 起叠加**（virtio-blk 先上，virtio-net 可选） |

结论写成一句工程判断：**先"完整 PC"把系统跑起来，再"virtio"把性能拉起来。** 反过来做（先 virtio）会因为"驱动进不了 initrd"卡死；只做前者（不做 virtio）会因为"IO 慢到无法使用"而放弃。

## 13.10 设备注册与冲突检测

```c
int vdev_init_all(vm_t *vm)
{
    static const struct { const vdev_ops_t *ops; u16 io; u16 iolen; u32 mmio; u32 mmiolen; u32 irq; } tbl[] = {
        { &vdev_pic_ops,  0x0020, 2,  0, 0, VDEV_IRQ_NONE },
        { &vdev_pic_ops,  0x00A0, 2,  0, 0, VDEV_IRQ_NONE },   /* index=1 从片 */
        { &vdev_pit_ops,  0x0040, 4,  0, 0, 0 },
        { &vdev_uart_ops, 0x03F8, 8,  0, 0, 4 },
        { &vdev_kbd_ops,  0x0060, 1,  0, 0, 1 },
        { &vdev_pci_ops,  0x0CF8, 8,  0, 0, VDEV_IRQ_NONE },
        { &vdev_vbe_ops,  0x01CE, 2,  0xF0000000u, 8u << 20, VDEV_IRQ_NONE },
    };

    for (u32 i = 0; i < sizeof(tbl)/sizeof(tbl[0]); i++) {
        vdev_t *d = &g_vdev_pool[vm->dev_cnt];
        memset(d, 0, sizeof(*d));
        d->ops = tbl[i].ops;
        d->io_base = tbl[i].io;
        d->io_len  = tbl[i].iolen;
        d->mmio_base = tbl[i].mmio;
        d->mmio_len  = tbl[i].mmiolen;
        d->irq = tbl[i].irq;
        d->vm = vm;
        d->index = vdev_next_index(vm, d->ops);

        /* 冲突检测：端口范围与 MMIO 范围都要查 */
        for (u32 j = 0; j < vm->dev_cnt; j++) {
            vdev_t *o = vm->devs[j];
            if (o->io_len && d->io_len &&
                !(d->io_base + d->io_len <= o->io_base || o->io_base + o->io_len <= d->io_base)) {
                serial_puts("VDEV conflict(io): ");
                serial_puts(d->ops->name); serial_putc(' ');
                serial_puts(o->ops->name); serial_putc('\n');
                return -EBUSY;
            }
        }

        if (d->ops->init(d, vm) < 0) return -EIO;
        vm->devs[vm->dev_cnt++] = d;

        /* 把端口范围在 IOPM 里放行 */
        if (d->io_len) iopm_allow(vm, d->io_base, d->io_len);
    }
    return 0;
}
```

注意最后一步：**注册设备的同时放行端口**，而不是逐设备手工 `iopm_allow`。这样"忘了放行导致 guest 收到 `#GP`"这类错误在结构上被消除。
