# 第 11 章 阶段 1：最小闭环的完整代码骨架

阶段 1 的目标只有一句话：**让 cunos 里跑起一段不是 cunos 的代码，并且它的输出出现在宿主串口上。** 这是整个项目的第一个"真机现象"，也是所有偏移/层级假设的试金石。

## 11.1 guest：16 位引导扇区（汇编源）

guest 不能调用 BIOS（我们没提供），所以它唯一能做的事就是**直接 `out` 到 COM1 数据端口**——这恰好也验证了 IOIO 退出路径。

```asm
; guest/hello16.asm
; nasm -f bin hello16.asm -o hello16.bin
BITS 16
ORG 0x7C00

_start:
    cli                     ; 不依赖中断
    cld                     ; lodsb 需要 DF=0（宿主已置，双保险）
    mov si, msg
.next:
    lodsb
    or  al, al
    jz  .hang
    mov dx, 0x3F8           ; COM1 数据寄存器
    out dx, al
    jmp .next
.hang:
    mov dx, 0x3F8 + 5       ; 线路状态寄存器
.wait:
    in  al, dx              ; 读一次：验证 IN 方向的 IOIO 退出
    test al, 0x20           ; THRE
    jz  .wait
    hlt                     ; 触发 HLT 退出
    jmp .hang

msg: db 'GUEST: hello from cunos VMM', 13, 10, 0
times 510-($-$$) db 0
dw 0xAA55
```

这段代码有四个刻意的设计：

1. **只用 `out`/`in`/`hlt` 三类会退出的指令**——它们分别验证 IOIO 的两个方向与 HLT 退出，覆盖阶段 1 需要的全部退出码。
2. **不写 `0xAA55` 之外的引导约定**（不用 BIOS 传的 DL、不用 `int 13h`），因为它不是被 BIOS 加载的，是被宿主 `memcpy` 进去的。
3. **输出字符串以 `\r\n\0` 结尾**，方便宿主串口终端正确换行。
4. **最后 `hlt` + `jmp` 死循环**，保证即使 HLT 处理没生效，guest 也不会跑飞。

## 11.2 宿主：把 guest 装进 RAM 并跑起来（约 40 行）

```c
/* 由构建脚本用 xxd -i 生成的数组；也可在 Makefile 里 objcopy 成 .o */
extern const unsigned char _binary_hello16_bin_start[];
extern const unsigned char _binary_hello16_bin_end[];

int vm_selftest_hello(void)
{
    if (vmm_init() < 0) { serial_puts("VMM init failed\n"); return -1; }

    vm_t *vm = vm_create("hello", 32u << 20);      /* 32MB guest RAM */
    if (!vm) return -1;

    /* 1) 把 guest 代码搬进 guest 物理地址 0x7C00 */
    u32 len = (u32)(_binary_hello16_bin_end - _binary_hello16_bin_start);
    if (len > 512) { serial_puts("guest too big\n"); return -1; }
    memcpy(ram_va(0x7C00), _binary_hello16_bin_start, len);

    /* 2) 构造 guest 初始上下文（第 10.4 节） */
    vcpu_t *vc = vm->vcpus;
    if (vcpu_init_real16(vc, 0x7C00) < 0) return -1;
    vc->exit_on_hlt = 1;                            /* hello 跑完就回宿主 */

    /* 3) 打开 UART 端口（IOPM 对应位清 0 表示"放行给设备模型"） */
    iopm_allow(vm, 0x3F8, 8);                       /* COM1: 0x3F8-0x3FF */

    /* 4) 跑 */
    serial_puts("[host] entering guest...\n");
    int r = vm_run(vm);
    serial_puts("[host] guest exited, r=");
    serial_dec((u32)r);
    serial_puts(" reasons: exits=");   serial_dec((u32)vc->exit_count);
    serial_puts(" io=");              serial_dec((u32)vc->io_count);
    serial_puts(" hlt=");             serial_dec((u32)vc->hlt_count);
    serial_putc('\n');
    return r;
}
```

`iopm_allow()` 的实现（注意位序：bit0 = 该字节的第一个端口）：

```c
void iopm_allow(vm_t *vm, u16 port, u32 count)
{
    u8 *b = (u8 *)ctl_va(vm->iopm_pa);
    for (u32 i = 0; i < count; i++) {
        u32 p = port + i;
        b[p >> 3] &= (u8)~(1u << (p & 7));    /* 清 0 = 不拦截 */
    }
}
```

## 11.3 五分钟实验：确认 NPT 的层级假设

在做完 11.2 之后如果**没有出现 `GUEST: hello`**，第一个要排除的就是 6.2 提到的 NPT 层级假设。专用实验如下（在 `vm_selftest_hello()` 之前插入）：

```c
/* 手工搭一个"最小 4 级 NPT"：只映射 GPA 0 开始的 2MB，
 * 用来判定硬件是否按 4 级 walk（PML4→PDP→PD→PT）。 */
int npt_probe_levels(void)
{
    u32 pml4_pa = ctl_alloc_page();
    u32 pdp_pa  = ctl_alloc_page();
    u32 pd_pa   = ctl_alloc_page();
    u32 pt_pa   = ctl_alloc_page();
    if (!pml4_pa || !pdp_pa || !pd_pa || !pt_pa) return -1;

    u64 *pml4 = ctl_va(pml4_pa), *pdp = ctl_va(pdp_pa);
    u64 *pd   = ctl_va(pd_pa),   *pt  = ctl_va(pt_pa);

    /* GPA 0..2MB 用 4KB 页铺到 HPA VMM_RAM_PA_BASE..+2MB */
    for (u32 i = 0; i < 512; i++)
        pt[i] = (u64)(VMM_RAM_PA_BASE + i * 0x1000u) | 0x007ull;   /* P|RW|US */
    pd[0]   = (u64)pt_pa   | 0x007ull;
    pdp[0]  = (u64)pd_pa   | 0x007ull;
    pml4[0] = (u64)pdp_pa  | 0x007ull;

    /* 把 guest 的 0x7C00 填成一条 hlt（0xF4），然后 VMRUN，
     * 若退出码为 HLT → 4 级假设成立；若为 NPF(0x400) 且 exit_info_2 为 0
     * → 层级被折叠，需按实测重建（见 6.2）。 */
    *(u8 *)ram_va(0x7C00) = 0xF4;
    return npt_probe_run(pml4_pa);
}
```

执行完这一步，把 `npt_probe_run()` 打印的 `exit_code` 抄进项目日志。**这个数字决定了 `npt.c` 的全部索引逻辑，值得单独留一条记录。**

## 11.4 VMRUN 失败检测

`VMRUN` 失败不会产生 `#VMEXIT`，表现为宿主收到 `#GP`/`#UD`。因此必须在 VMM 初始化时挂钩 IDT：

```c
/* idt.c 需新增：允许替换某个向量的处理函数 */
void idt_set_handler(u32 vec, void (*fn)(int_regs_t *));

static void vmm_gp_hook(int_regs_t *r)
{
    if (g_in_vmrun) {
        serial_puts("\n[VMRUN FAILED] #GP in VMRUN, eip=0x");
        serial_hex(r->eip);
        serial_puts(" err=0x"); serial_hex(r->err_code);
        serial_puts("\ncheck: EFER.SVME=1 / VMCB 4K-aligned / HSAVE set / NCR3 set\n");
        vmm_panic("VMRUN #GP", r->eip, r->err_code);
    }
    /* 否则是 guest 的 #GP 被注入到宿主？不可能——guest 的 #GP 不会走宿主 IDT。
     * 走到这里的 true 分支说明宿主自己执行了非法指令，按普通异常处理。 */
    panic_gp(r);
}

static void vmm_ud_hook(int_regs_t *r)
{
    if (g_in_vmrun) vmm_panic("VMRUN #UD: SVM not enabled on this CPU", r->eip, 0);
    panic_ud(r);
}
```

**这两个钩子是"阶段 1 能不能推进"的关键**。没有它们，VMRUN 失败的表现是"屏幕静止 / 立刻重启（三落）"，完全无法定位。

## 11.5 主循环完整骨架

```c
#define VCPU_EXIT_NORMAL    0
#define VCPU_EXIT_SHUTDOWN  1

int vm_run(vm_t *vm)
{
    vcpu_t *vc = vm->vcpus;
    vm->state = VM_STATE_RUNNING;
    vc->state = VCPU_STATE_RUNNING;

    for (;;) {
        vcpu_enter_guest(vc);                     /* VMRUN，见 5.6 */

        u32 code = vmcb_rd32(vc->vmcb, VMCB_OFF_EXIT_CODE) & 0xFFFFu;
        vc->exit_count++;
        vc->last_exit_code = code;
        vc->last_rip = (u32)vmcb_rd64(vc->vmcb, VMCB_OFF_RIP);
        vmexit_trace_push(code, vc->last_rip);     /* 第 18 章 */

        /* 每次退出都是"时间前进"的机会：让设备检查到期事件 */
        u64 now = rdtsc();
        for (u32 i = 0; i < vm->dev_cnt; i++)
            if (vm->devs[i]->ops->tick)
                vm->devs[i]->ops->tick(vm->devs[i], now);

        /* 递送待注入事件（PIT tick / UART THRE 等） */
        vcpu_deliver_pending(vc);

        int r = vmexit_dispatch(vc, code);
        if (r == VCPU_EXIT_NORMAL)   { vm->state = VM_STATE_STOPPED; return r; }
        if (r == VCPU_EXIT_SHUTDOWN) { vm->state = VM_STATE_STOPPED; return r; }
        if (r < 0)                   { vm->state = VM_STATE_DEAD;    return r; }
    }
}
```

## 11.6 阶段 1 验收标准

全部为**可观察现象**，逐条打勾：

| # | 验收项 | 期望现象 | 判据 |
|---|---|---|---|
| 1 | SVM 可用性 | 宿主串口打印 `SVM: rev=... nasid=... npt=1` | 数字与 CPU 型号可交叉验证（`NASID` 通常 ≤ 64） |
| 2 | EFER 使能 | 打印 `SVM: enabled, EFER=0x1...`（含 bit12） | `(efer & 0x1000) != 0` |
| 3 | HSAVE | 打印 `SVM: HSAVE=0x03xxxxxx` | 地址落在控制池 `0x03xxxxxx`，且 4KB 对齐 |
| 4 | NPT | 打印 `NPT: ram=32MB ncr3=0x03xxxxxx` | `ncr3` 4KB 对齐 |
| 5 | **进入 guest** | 打印 `[host] entering guest...` 后**宿主串口出现** `GUEST: hello from cunos VMM` | 这是第一个"非 cunos 代码产生输出"的证据 |
| 6 | IOIO 退出 | `io=` 计数 ≥ 30（27 个字符 + 若干 CR/LF + THRE 轮询） | 与字符串长度一致 |
| 7 | HLT 退出 | `hlt=1`，随后打印 `[host] guest exited, r=0` | HLT 被正确拦截，宿主没有挂死 |
| 8 | 无三落 | 机器不重启，shell 可继续执行下一条命令 | 手工确认 |
| 9 | 偏移正确性 | 无 `NPF` 退出，无 `#GP` 注入 | `npf=0` |

**第 5 条是阶段 1 的唯一硬指标**：只要它出现，说明 CPUID 探测、EFER、HSAVE、VMCB 布局、NPT 层级、IOPM、IOIO 分发、HLT 拦截七件事全部正确。

## 11.7 失败现象 → 病因对照表

这张表是本文最实用的部分，建议打印贴在显示器边。

| 现象 | 最可能的病因 | 排查动作 |
|---|---|---|
| `[host] entering guest...` 之后**完全无输出且机器重启** | VMRUN 失败后三落（EFER 未开 / VMCB 未 4KB 对齐 / HSAVE 未设） | 加 11.4 的钩子；先确认 `rdmsr(EFER)` 的 bit12 |
| 进入 guest 后立刻退出，`exit_code = 0x400`，`exit_info_2 = 0` | NPT 未映射 GPA 0（或层级折叠） | 跑 11.3 实验；检查 `ncr3` 是否真的写进 VMCB（读回来比对） |
| `exit_code = 0x400`，`exit_info_2 = 0x7C00` | guest RAM 起点的映射缺失（`npt_build` 的循环边界错） | 打印 `npt_walk(0x7C00)` 的完整结果 |
| 输出的是**乱码或宿主自己的字符串** | `ram_va()` 窗口映射错位（`paging_map_page` 的 `g_pd` bug） | 直接比对 `*(u32*)ram_va(0x7C00)` 与 guest 二进制前 4 字节 |
| 串口一个字符打印，然后 `exit_code` 恒为 `0x073` 且 `RIP` 不变 | RIP 没有前推（谁负责推：IOIO 处理里 `RIP += 指令长度`） | 打印每次的 RIP，看是否递增 |
| 打印到一半停住，`exit_code = 0x070`（HLT），宿主也卡住 | `exit_on_hlt` 未置 1，主循环又回到 VMRUN，而 guest 已 HLT（不会再有退出） | 检查 `exit_on_hlt` 与 HLT 处理分支 |
| `exit_code` 恒为 `0x006` 或 `0x00D`（异常被拦） | `intercept_exceptions` 误开（阶段 1 应全 0） | 清零该字段 |
| VMRUN 成功但 `RIP` 停在 0 反复退出 | `CS` 的 base/selector 与 `RIP` 不匹配（实模式 base 应为 0） | 读回 VMCB 的 CS 槽位 |
| 一切正常但第二次运行 `vm hello` 崩溃 | 控制池 bump 耗尽 / 重复 `vmm_init` | 加 `g_vmm_inited` 幂等保护 |
