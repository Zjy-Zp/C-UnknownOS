# 第 9 章 核心数据结构

设计原则三条，先立后写：

1. **硬件结构体只用"字节数组 + 偏移访问器"**，不用大结构体成员。理由见 4.5.2。
2. **软件状态与硬件状态分离**。`vcpu_t` 存"宿主视角的状态"（计数器、状态机、pending 中断），VMCB 存"guest 视角的 CPU 状态"。两者只在 `VMRUN` 交界处一致。
3. **所有对象都在静态池里，不 `malloc`**。cunos 无 `free`，所以用固定数组 + bump 分配，`VMM_MAX_VMS = 4`、`VMM_MAX_VCPUS = 8`。

## 9.1 `vdev_t`：设备模型统一接口

先把设备接口定下来，因为 `vm_t` 要持有设备数组。

```c
/* vmm/vdev.h */
typedef struct vm     vm_t;
typedef struct vdev   vdev_t;

typedef struct {
    const char *name;

    /* 初始化：把自己的端口范围/MMIO 范围/IRQ 填进 vdev，并复位寄存器堆 */
    int (*init)(vdev_t *d, vm_t *vm);

    /* 端口 IO：width ∈ {1,2,4}，返回 0 表示已处理，<0 表示未实现（会注入 #GP） */
    int (*io_read)(vdev_t *d, u16 port, u32 width, u32 *out);
    int (*io_write)(vdev_t *d, u16 port, u32 width, u32 val);

    /* MMIO：gpa 落在本设备的 mmio 范围内时被调用。
     * out_pa != 0 表示"把这段 GPA 直接映射到该 HPA"（用于帧缓冲直通）。 */
    int (*mmio_fault)(vdev_t *d, u32 gpa, u32 err, u32 *out_pa);

    /* 时间推进：每个退出点被调用一次，用于 PIT/UART 的到期检查 */
    int (*tick)(vdev_t *d, u64 now_tsc);

    /* 复位（阶段 4 的 reboot 支持） */
    int (*reset)(vdev_t *d);
} vdev_ops_t;

struct vdev {
    const vdev_ops_t *ops;
    u16  io_base;
    u16  io_len;
    u32  mmio_base;
    u32  mmio_len;
    u32  irq;            /* 999 = 未连线 */
    u32  index;          /* 同类型设备序号（如 uart0/uart1） */
    void *priv;          /* 设备私有状态（寄存器堆等） */
    vm_t *vm;
};

#define VDEV_IRQ_NONE 999
#define VDEV_MAX      16
```

## 9.3 `vcpu_t`

```c
#define VCPU_STATE_INIT      0   /* 刚分配，上下文未填 */
#define VCPU_STATE_READY     1   /* 上下文就绪，可 VMRUN */
#define VCPU_STATE_RUNNING   2   /* 已进入 VMRUN 阶段（宿主视角） */
#define VCPU_STATE_HALTED    3   /* guest 执行了 hlt，等待事件 */
#define VCPU_STATE_SHUTDOWN  4   /* guest 三落/关机 */
#define VCPU_STATE_FAULTED   5   /* 不可恢复错误，宿主侧问题 */

typedef struct {
    u32      id;
    u32      state;
    vm_t    *vm;

    u32      vmcb_pa;        /* VMCB 物理地址（4KB 对齐） */
    u8      *vmcb;           /* VMCB 宿主虚拟地址（控制窗口） */
    u32      hsave_pa;       /* 本 vCPU 专属 HSAVE 页 */

    u32      exit_on_hlt;    /* 1 = hlt 时结束 vm_run（阶段 1 用） */
    u32      halt_wake_tick; /* HALTED 状态下的唤醒时刻（TSC 刻度） */

    /* 统计（第 18 章用） */
    u64      exit_count;
    u64      io_count;
    u64      msr_count;
    u64      npf_count;
    u64      cpuid_count;
    u64      hlt_count;
    u32      last_exit_code;
    u32      last_rip;

    /* 待注入事件（由设备设置，由 vmexit 消费） */
    u32      pending_vector;
    u32      pending_edge;   /* 1 = 边沿触发型注入（PIC 的 IRQ 上升沿） */
} vcpu_t;
```

字段注释里的取舍说明：

- 没有"guest 寄存器缓存"字段。因为所有 guest 寄存器的权威副本在 VMCB 里，多存一份就会出现"两份不一致"的经典 bug。
- `halt_wake_tick` 用 TSC 计数而不是"次数"，因为 tick 的频率依赖 PIT 编程值，用绝对时刻更好推理。

## 9.4 `vm_t`

```c
#define VM_STATE_NEW      0
#define VM_STATE_READY    1
#define VM_STATE_RUNNING  2
#define VM_STATE_STOPPED  3
#define VM_STATE_DEAD     4

struct vm {
    u32      id;
    char     name[16];
    u32      state;

    /* 内存 */
    u32      ram_gpa_base;     /* 恒为 0（本文不支持非零起始的 RAM） */
    u32      ram_size;         /* 实际给 guest 的 RAM 字节数（2MB 对齐） */
    npt_t    npt;

    /* 控制结构 */
    u32      iopm_pa;          /* 12KB：3 张 4KB 页 */
    u32      msrpm_pa;         /* 12KB */
    u32      vcpu_cnt;
    vcpu_t  *vcpus;            /* 指向静态池，长度 vcpu_cnt */

    /* 设备 */
    u32      dev_cnt;
    vdev_t  *devs[VDEV_MAX];

    /* 时间 */
    u64      tsc_base;         /* guest TSC = 宿主 TSC - tsc_base（阶段 4） */

    u32      shutdown_reason;
};
```

## 9.5 VMCB 访问器：把偏移表变成 API

```c
/* vmm/vmcb.h */
#define VMCB_SIZE            4096u

/* control area */
#define VMCB_OFF_ICR_READ    0x000u
#define VMCB_OFF_ICR_WRITE   0x002u
#define VMCB_OFF_IDR_READ    0x004u
#define VMCB_OFF_IDR_WRITE   0x006u
#define VMCB_OFF_IEXCP       0x008u
#define VMCB_OFF_INTERCEPT   0x00Cu
#define VMCB_OFF_IOPM_PA     0x040u
#define VMCB_OFF_MSRPM_PA    0x048u
#define VMCB_OFF_TSC_OFFSET  0x050u
#define VMCB_OFF_ASID        0x058u
#define VMCB_OFF_TLB_CTL     0x05Cu
#define VMCB_OFF_INT_CTL     0x060u
#define VMCB_OFF_INT_VECTOR  0x064u
#define VMCB_OFF_EXIT_CODE   0x070u
#define VMCB_OFF_EXIT_INFO1  0x078u
#define VMCB_OFF_EXIT_INFO2  0x080u
#define VMCB_OFF_EXIT_INTINFO 0x088u
#define VMCB_OFF_NCR3        0x0A8u      /* 【需核对】 */
#define VMCB_OFF_CLEAN       0x0C8u      /* 【需核对】 */

/* state save area */
#define VMCB_OFF_ES          0x400u
#define VMCB_OFF_CS          0x410u
#define VMCB_OFF_SS          0x420u
#define VMCB_OFF_DS          0x430u
#define VMCB_OFF_FS          0x440u
#define VMCB_OFF_GS          0x450u
#define VMCB_OFF_GDTR        0x460u
#define VMCB_OFF_LDTR        0x470u
#define VMCB_OFF_IDTR        0x480u
#define VMCB_OFF_TR          0x490u
#define VMCB_OFF_CPL         0x4B0u      /* 【需核对】 */
#define VMCB_OFF_EFER        0x4D0u
#define VMCB_OFF_CR4         0x4E0u
#define VMCB_OFF_CR3         0x4E8u
#define VMCB_OFF_CR0         0x4F0u
#define VMCB_OFF_DR7         0x4F8u
#define VMCB_OFF_DR6         0x500u
#define VMCB_OFF_RFLAGS      0x508u
#define VMCB_OFF_RIP         0x510u
#define VMCB_OFF_RSP         0x520u
#define VMCB_OFF_RAX         0x530u
#define VMCB_OFF_CR2         0x578u

static inline u32  vmcb_rd32(u8 *v, u32 off)  { return *(volatile u32 *)(v + off); }
static inline void vmcb_wr32(u8 *v, u32 off, u32 x) { *(volatile u32 *)(v + off) = x; }
static inline u16  vmcb_rd16(u8 *v, u32 off)  { return *(volatile u16 *)(v + off); }
static inline void vmcb_wr16(u8 *v, u32 off, u16 x) { *(volatile u16 *)(v + off) = x; }
static inline u64  vmcb_rd64(u8 *v, u32 off)  { return *(volatile u64 *)(v + off); }
static inline void vmcb_wr64(u8 *v, u32 off, u64 x) { *(volatile u64 *)(v + off) = x; }

/* 32 位宿主对 guest 通用寄存器的访问：只取/写低 32 位，高 32 位清零 */
static inline u32 vmcb_get_eax(u8 *v) { return vmcb_rd32(v, VMCB_OFF_RAX); }
static inline void vmcb_set_eax(u8 *v, u32 x) { vmcb_wr32(v, VMCB_OFF_RAX, x); vmcb_wr32(v, VMCB_OFF_RAX + 4, 0); }

/* 段寄存器 16 字节结构 */
typedef struct {
    u16 selector;   /* +0 */
    u16 attrib;     /* +2 */
    u32 limit;      /* +4 */
    u32 base;       /* +8（32 位模式只用低 32 位） */
    u32 base_hi;    /* +12（保留，置 0） */
} vmcb_seg32_t;

static inline void vmcb_set_seg(u8 *v, u32 off, u16 sel, u16 attrib, u32 limit, u32 base)
{
    vmcb_wr16(v, off + 0, sel);
    vmcb_wr16(v, off + 2, attrib);
    vmcb_wr32(v, off + 4, limit);
    vmcb_wr32(v, off + 8, base);
    vmcb_wr32(v, off + 12, 0);
}
```

**偏移正确性的自检手段**（没有 gdb 的情况下）：阶段 1 的 hello guest 只用得到 `RIP`/`RSP`/`CS`/`GDTR`/`CR0`/`RFLAGS` 七个字段。做法是**逐个字段启用**：先只填 RIP/CS/CR0 让 guest 跑一条 `hlt`，确认能退出；再加 RSP；再加 GDTR。哪一步加进去就崩，问题就锁定在那个偏移上。这是本章最重要的实操方法，比"一次性填满然后赌"快一个数量级。
